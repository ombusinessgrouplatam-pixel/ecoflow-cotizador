(Lee el Google Sheet desde el servidor — seguro)

```js
export async function GET() {
  const sheetId = process.env.NEXT_PUBLIC_SHEET_ID;
  const url = `https://docs.google.com/spreadsheets/d/${sheetId}/export?format=csv`;

  try {
    const res = await fetch(url, { next: { revalidate: 60 } });
    if (!res.ok) throw new Error("Sheet error");
    const text = await res.text();
    return Response.json({ csv: text });
  } catch {
    return Response.json({ csv: null }, { status: 500 });
  }
}
```

```js
export async function POST(req) {
  const { base64, mediaType } = await req.json();
  const isPDF = mediaType === "application/pdf";

  const block = isPDF
    ? { type: "document", source: { type: "base64", media_type: "application/pdf", data: base64 } }
    : { type: "image", source: { type: "base64", media_type: mediaType, data: base64 } };

  const res = await fetch("https://api.anthropic.com/v1/messages", {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "x-api-key": process.env.ANTHROPIC_API_KEY,
      "anthropic-version": "2023-06-01",
    },
    body: JSON.stringify({
      model: "claude-sonnet-4-20250514",
      max_tokens: 256,
      messages: [{
        role: "user",
        content: [
          block,
          { type: "text", text: "Analiza este recibo de energía eléctrica. Extrae la potencia demandada en watts (W) o kW. Si es kW multiplica por 1000. Responde SOLO con un número entero. Si no puedes determinarlo responde: 0" }
        ]
      }]
    })
  });

  const data = await res.json();
  const watts = parseInt(data.content?.[0]?.text?.trim() || "0", 10) || 0;
  return Response.json({ watts });
}
```

```jsx
function findBest(data, w) {

function parseCSV(text) {
  const lines = text.trim().split("\n").filter(l => l.trim());
  const hi = lines.findIndex(l => l.toLowerCase().includes("referencia"));
  if (hi === -1) return null;
  const headers = lines[hi].split(",").map(h => h.replace(/"/g,"").trim());
  const rows = lines.slice(hi+1).map(line => {
    const vals = []; let cur = "", inQ = false;
    for (const c of line) {
      if (c==='"') inQ=!inQ;
      else if (c===',' && !inQ) { vals.push(cur.trim()); cur=""; }
      else cur+=c;
    }
    vals.push(cur.trim());
    const obj = {};
    headers.forEach((h,i) => { obj[h] = (vals[i]||"").replace(/[$,]/g,"").trim(); });
    return obj;
  }).filter(r => r[headers[0]] && r[headers[0]].length > 2);

  return rows.map(r => ({
    ref: r["Referencia equipo"],
    cap: Number(r["Capacidad (Wh)"]) || 0,
    pot: Number(r["Potencia (W)"]) || 0,
    panel: r["Panel recomendado"] || "-",
    qty: Number(r["Cantidad paneles recomendados"]) || 0,
    precioPanel: Number(r["Precio Panel (COP) excepto IVA"]) || 0,
    precioSinIVA: Number(r["Precio Equipo EcoFlow antes de IVA"]) || 0,
    precioConIVA: Number(r["Precio Equipo EcoFlow incluido IVA"]) || 0,
  })).filter(r => r.ref && r.pot > 0);
}

export default function Home() {

  const [result, setResult] = useState(null);
  const [sheetErr, setSheetErr] = useState(false);
  const [ocrInfo, setOcrInfo] = useState(null);
  const [fileName, setFileName] = useState("");
  const fileRef = useRef();

  const getData = useCallback(async () => {
    try {
      const res = await fetch("/api/sheet");
      const { csv } = await res.json();
      if (!csv) throw new Error();
      const parsed = parseCSV(csv);
      if (!parsed || !parsed.length) throw new Error();
      return parsed;
    } catch {
      setSheetErr(true);
      return DEMO_DATA;
    }
  }, []);

  const search = async (w) => {
    const val = parseInt(w ?? watts, 10);
    if (!val || val <= 0) return;
    setLoading(true); setResult(null);
    const data = await getData();
    setResult(findBest(data, val));
    setLoading(false);
  };

  const handleFile = async (file) => {
    if (!file) return;
    setFileName(file.name); setOcrLoading(true); setOcrInfo(null);
    try {
      const b64 = await new Promise((res,rej)=>{
        const r=new FileReader();
        r.onload=()=>res(r.result.split(",")[1]);
        r.onerror=rej;
        r.readAsDataURL(file);
      });
      const res = await fetch("/api/ocr", {
        method:"POST",
        headers:{"Content-Type":"application/json"},
        body:JSON.stringify({base64:b64,mediaType:file.type})
      });
      const { watts: w } = await res.json();
      setOcrInfo(w); setWatts(String(w));
      if (w>0) await search(w);
    } catch { setOcrInfo(0); }
    setOcrLoading(false);
  };

            Encuentra tu solución<br/><span style={{color:"#555"}}>de energía ideal</span>
```
